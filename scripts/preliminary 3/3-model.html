  # calculates the tf values
  filter(.id %in% validation$.id) %>%
  count(.id, term) %>% 
  group_by(.id) %>%
  mutate(tf_2 = n / sum(n)) %>%
  ungroup() %>%

  # we only want to use words that are also in our training data
  inner_join(claims_tf_idf %>% 
              group_by(term) %>%
              slice_head(n=1) %>%
              select(term, idf),
            by = "term") %>%
  
  # calculates the tf_idf values 
  mutate(tf_idf = tf_2 * idf) %>%
  pivot_wider(id_cols = .id,
              names_from = term,
              values_from = tf_idf,
              values_fill = 0)
